```python
#!/usr/bin/env python3
"""Decorrelation check for a candidate blend member (numbers only: no ids, no text).

Per finding: AUC of A, AUC of B, AUC of the rank blend (1-w)*A + w*B, within-class Spearman of A vs B
(positives and negatives separately), and on positive-negative pairs: fix = share of the pairs A orders
wrongly that B orders rightly, brk = share of the pairs A orders rightly that B orders wrongly.
A useful new member has rho well below ~0.85 and fix clearly above brk.

    .venv/bin/python -I tools/decorr.py --a 'raptor_oof_oof4_f*_swa_fold*.npz' \
        --b 'pc192/raptor_oof_cnxt_f*_swa_fold*.npz' --truth labels_avg.parquet
    --truth gold:ds/comp/train.csv   -> the 58 expert-labelled studies (Report column never read)
"""
import argparse
import glob

import numpy as np
import pandas as pd
from scipy.stats import rankdata, spearmanr
from sklearn.metrics import roc_auc_score

LAB = ['ACL', 'MCL', 'Medial Meniscus', 'Lateral Meniscus', 'Medial OA', 'Lateral OA', 'PF OA',
       'Effusion', 'Synovitis', "Baker's", 'Contusion', 'Fracture']


def _table(path):
    df = pd.read_parquet(path) if path.endswith(".parquet") else pd.read_csv(
        path, usecols=["StudyInstanceUID"] + LAB, dtype={"StudyInstanceUID": str})
    return df.set_index(df["StudyInstanceUID"].astype(str))[LAB]


def load_preds(spec):
    files = sorted(glob.glob(spec))
    if not files:
        raise SystemExit(f"no files match {spec}")
    if files[0].endswith(".npz"):
        parts = []
        for f in files:
            z = np.load(f, allow_pickle=True)
            parts.append(pd.DataFrame(z["pred"], columns=LAB, index=z["ids"].astype(str)))
        df = pd.concat(parts)
    else:
        df = _table(files[0])
    if df.index.duplicated().any():
        raise SystemExit(f"{spec}: duplicate study ids")
    return df


def load_truth(spec):
    if spec.startswith("gold:"):
        return _table(spec[5:]).dropna()          # gold rows are the only fully labelled train rows
    return _table(spec)


def _pairs(y, a, b):
    d_a = a[y == 1][:, None] - a[y == 0][None, :]
    d_b = b[y == 1][:, None] - b[y == 0][None, :]
    wrong, right = d_a < 0, d_a > 0
    fix = float((d_b[wrong] > 0).mean()) if wrong.any() else 0.0
    brk = float((d_b[right] < 0).mean()) if right.any() else 0.0
    return fix, brk


def compare(A, B, T, w=0.3):
    ids = A.index.intersection(B.index).intersection(T.index)
    if len(ids) == 0:
        raise SystemExit("no common study ids between --a, --b and --truth")
    rows = []
    for c in LAB:
        y = (T.loc[ids, c].values >= 0.5).astype(int)
        if y.min() == y.max():
            continue
        a = rankdata(A.loc[ids, c].values) / len(ids)
        b = rankdata(B.loc[ids, c].values) / len(ids)
        fix, brk = _pairs(y, a, b)
        rows.append(dict(finding=c, n_pos=int(y.sum()),
                         auc_a=roc_auc_score(y, a), auc_b=roc_auc_score(y, b),
                         auc_blend=roc_auc_score(y, (1 - w) * a + w * b),
                         rho_pos=spearmanr(a[y == 1], b[y == 1])[0],
                         rho_neg=spearmanr(a[y == 0], b[y == 0])[0], fix=fix, brk=brk))
    return pd.DataFrame(rows).set_index("finding"), len(ids)


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--a", required=True)
    ap.add_argument("--b", required=True)
    ap.add_argument("--truth", required=True)
    ap.add_argument("--w", type=float, default=0.3)
    a = ap.parse_args()
    r, n = compare(load_preds(a.a), load_preds(a.b), load_truth(a.truth), a.w)
    pd.set_option("display.width", 160)
    print(f"common studies: {n} | blend weight on B: {a.w}")
    print(r.round(4).to_string())
    print("MACRO", r.drop(columns="n_pos").mean().round(4).to_dict())


if __name__ == "__main__":
    main()
```

def _study_inputs(model, vol, mask, res, dev):
    """Windows of one study for `model`'s layout, on `dev`: (x, slots_t, win_centers)."""
    roi_cfg = None
    if getattr(model, "roi", False):
        roi_cfg = {"mode": getattr(model, "roi_mode", "compartment"),
                   "pad": float(getattr(model, "roi_pad", 0.06)),
                   "overlap": float(getattr(model, "roi_overlap", 0.12))}
    x, slot_ids, win_centers = _build_windows(
        vol, mask, res,
        clean_win=getattr(model, "clean_windows", False),
        slots=getattr(model, "slots", None),
        k=K_EVAL, roi=roi_cfg,
        norm=getattr(model, "input_norm", "none"))
    if getattr(model, "seq_head", False):
        assert all(b >= a for a, b in zip(win_centers, win_centers[1:])), \
            "seq_head requires ascending window centers"
    x = x.to(dev, non_blocking=True)
    slots_t = torch.tensor(slot_ids, dtype=torch.long, device=dev)[None] if getattr(model, 'has_slot_emb', False) else None
    return x, slots_t, win_centers


def _probs(model, x, slots_t, win_centers, dev, use_tta):
    autocast_on = str(dev).startswith('cuda') and torch.cuda.is_available()
    with torch.autocast('cuda', dtype=torch.float16, enabled=autocast_on):
        p_plain = torch.sigmoid(model(x[None], slots=slots_t).float())[0]
        if use_tta:
            x_m = mirror_tri(x, win_centers)
            p_mirr = torch.sigmoid(model(x_m[None], slots=slots_t).float())[0]
            p_final = 0.5 * (p_plain + p_mirr)
        else:
            p_final = p_plain
    return p_final.detach().cpu().numpy()


def predict_study(model, vol, mask, res, dev, use_tta=False):
    t_m0 = time.time()
    x, slots_t, win_centers = _study_inputs(model, vol, mask, res, dev)
    p_np = _probs(model, x, slots_t, win_centers, dev, use_tta)
    return p_np, time.time() - t_m0


def predict_study_multi(models, vol, mask, res, dev, use_tta=False):
    """Mean probability of same-layout models (check_same_layout); the windows are built once."""
    t_m0 = time.time()
    x, slots_t, win_centers = _study_inputs(models[0], vol, mask, res, dev)
    p_np = np.mean([_probs(m, x, slots_t, win_centers, dev, use_tta) for m in models], axis=0)
    return p_np, time.time() - t_m0

def check_same_layout(cks):
    """Checkpoints scored on one decode pass must share the window layout; a mixed set would
    silently feed one model another model's inputs."""
    ref = {k: cks[0].get(k) for k in LAYOUT_KEYS}

    return ref

### Task 9 (GATED: needs the user's explicit approval to read the Report column with a LOCAL model): severity-graded labels

Only run this if the user lifts the "never open the Report column" rule for a local-only pass. The research ranks it as the highest-value label idea (+0.003 to +0.010, by estimate). It targets the three OA findings, effusion, Baker's cyst and the meniscus definitions. Nothing is printed except counts, and nothing leaves the machine.

**Files:**
- Create: `tools/severity_grades.py`
- Output: `labels_sev_v1.parquet` (same format as `labels_avg.parquet`)

- [ ] **Step 1: Implement `tools/severity_grades.py`** (vLLM, `Qwen/Qwen2.5-7B-Instruct-AWQ` on the PC, or `Qwen/Qwen2.5-14B-Instruct` on an A100; both Apache-2.0)

```python
#!/usr/bin/env python3
"""Local-only severity grading of report mentions -> soft targets (prints counts only, never text or ids).

Grades only cells whose 31-table mean >= 0.2 (presence still comes from the consensus); the LLM only grades.
target = mean * G[grade], G fixed from the host rubric (never fitted on gold-58).
    .venv/bin/python -I tools/severity_grades.py --model Qwen/Qwen2.5-7B-Instruct-AWQ --out labels_sev_v1.parquet
"""
import argparse
import json
from collections import Counter

import pandas as pd

LAB = ['ACL', 'MCL', 'Medial Meniscus', 'Lateral Meniscus', 'Medial OA', 'Lateral OA', 'PF OA',
       'Effusion', 'Synovitis', "Baker's", 'Contusion', 'Fracture']
VOCAB = {  # grade -> target multiplier (>= host threshold ~0.95, below ~0.2, unclear 0.65)
    'ACL': {'intact': 0.1, 'low_grade_sprain': 0.2, 'high_grade_partial': 0.95, 'complete': 0.95, 'unclear': 0.65},
    'MCL': {'intact': 0.1, 'low_grade_sprain_or_chronic': 0.2, 'acute_high_grade_or_complete': 0.95, 'unclear': 0.65},
    'Medial Meniscus': {'normal': 0.1, 'intrasubstance_signal': 0.2, 'possible_tear': 0.5, 'definite_tear': 0.95, 'unclear': 0.65},
    'Lateral Meniscus': {'normal': 0.1, 'intrasubstance_signal': 0.2, 'possible_tear': 0.5, 'definite_tear': 0.95, 'unclear': 0.65},
    'Medial OA': {'none': 0.1, 'low_grade_or_small': 0.2, 'high_grade_large': 0.95, 'unclear': 0.65},
    'Lateral OA': {'none': 0.1, 'low_grade_or_small': 0.2, 'high_grade_large': 0.95, 'unclear': 0.65},
    'PF OA': {'none': 0.1, 'low_grade_or_small': 0.2, 'high_grade_large': 0.95, 'unclear': 0.65},
    'Effusion': {'none': 0.1, 'trace_or_small': 0.2, 'moderate_or_large': 0.95, 'unclear': 0.65},
    "Baker's": {'none': 0.1, 'small': 0.2, 'moderate_or_large': 0.95, 'unclear': 0.65},
    'Contusion': {'none': 0.1, 'degenerative_or_nonspecific': 0.2, 'traumatic_impact': 0.95, 'unclear': 0.65},
    'Fracture': {'none': 0.1, 'old_or_stress': 0.2, 'acute_fracture_line': 0.95, 'unclear': 0.65},
}
RUBRIC = ("Grade each listed knee finding from the radiology report using ONLY these words. Rules: OA counts as "
          "high_grade_large only for >50% cartilage-thickness loss over about 1 cm or more in that compartment; "
          "effusion/Baker's only moderate or large count; a meniscal tear needs definite surface-reaching signal or "
          "abnormal morphology; ACL needs complete or >50% fiber disruption; MCL needs an acute high-grade or complete "
          "tear; contusion is impact-pattern marrow edema without a fracture line; fracture needs an acute line. "
          "Answer with one JSON object mapping finding to grade.")


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument('--model', required=True)
    ap.add_argument('--train_csv', default='ds/comp/train.csv')
    ap.add_argument('--avg', default='labels_avg.parquet')
    ap.add_argument('--out', required=True)
    a = ap.parse_args()
    from vllm import LLM, SamplingParams
    avg = pd.read_parquet(a.avg).set_index('StudyInstanceUID')
    rep = pd.read_csv(a.train_csv, usecols=['StudyInstanceUID', 'Report'], dtype=str).set_index('StudyInstanceUID')['Report']
    llm = LLM(model=a.model, max_model_len=4096)
    tok = llm.get_tokenizer()
    jobs, prompts = [], []
    for uid in avg.index:
        todo = [c for c in VOCAB if avg.at[uid, c] >= 0.2]
        if not todo or uid not in rep.index or not isinstance(rep[uid], str):
            continue
        opts = {c: list(VOCAB[c]) for c in todo}
        msg = [{'role': 'system', 'content': RUBRIC},
               {'role': 'user', 'content': 'Findings and allowed grades: ' + json.dumps(opts) + '\n\nReport:\n' + rep[uid]}]
        prompts.append(tok.apply_chat_template(msg, tokenize=False, add_generation_prompt=True))
        jobs.append((uid, todo))
    outs = llm.generate(prompts, SamplingParams(temperature=0.0, max_tokens=300))
    sev, counts, bad = avg.copy(), Counter(), 0
    for (uid, todo), o in zip(jobs, outs):
        try:
            g = json.loads(o.outputs[0].text.strip().strip('`').removeprefix('json').strip())
        except Exception:
            bad += 1
            continue
        for c in todo:
            grade = g.get(c, 'unclear') if isinstance(g, dict) else 'unclear'
            mult = VOCAB[c].get(grade, 0.65)
            sev.at[uid, c] = float(avg.at[uid, c]) * mult
            counts[(c, grade if grade in VOCAB[c] else 'invalid')] += 1
    sev.reset_index().to_parquet(a.out, index=False)
    print(f'graded studies {len(jobs)} | unparsable {bad}')
    for c in VOCAB:
        print(c, {k[1]: v for k, v in sorted(counts.items()) if k[0] == c})


if __name__ == '__main__':
    main()
```
